cmake_minimum_required(VERSION 3.24)
project(job_system LANGUAGES CXX)
include(FetchContent)
include(CTest)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(jobs src/jobs.cpp)
target_include_directories(jobs PUBLIC include)
target_link_libraries(jobs PUBLIC Threads::Threads)
add_executable(jobs_demo demo.cpp)
target_link_libraries(jobs_demo PRIVATE jobs)
if(BUILD_TESTING)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_Declare(googletest URL https://codeload.github.com/google/googletest/tar.gz/refs/tags/v1.16.0)
    FetchContent_MakeAvailable(googletest)
    add_executable(jobs_tests tests.cpp)
    target_link_libraries(jobs_tests PRIVATE jobs GTest::gtest_main)
    add_test(NAME jobs_tests COMMAND jobs_tests)
endif()
